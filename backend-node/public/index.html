// Node.js / Express Backend Verification
const express = require('express');
const app = express();
app.use(express.json());

const CAPTCHA_SERVER = 'http://localhost:3000';
const SITE_SECRET = process.env.SITE_SECRET || 'sec_shield_live_...';

app.post('/api/login', async (req, res) => {
  const { email, captchaToken } = req.body;

  // 1. Authorize token with ShieldCaptcha Server
  const verifyRes = await fetch(`${CAPTCHA_SERVER}/api/siteverify`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${SITE_SECRET}`
    },
    body: JSON.stringify({
      token: captchaToken,
      ip: req.socket.remoteAddress // Cryptographic IP binding
    })
  });

  const { success, score, error } = await verifyRes.json();
  if (!success || score < 30) {
    return res.status(403).json({ error: 'CAPTCHA authorization failed: ' + error });
  }

  // 2. Token is authorized & consumed (Single-Use guaranteed)!
  res.json({ success: true, message: `Welcome ${email}! (Trust Score: ${score})` });
});

app.listen(8080);

// Go (Golang) Backend Verification
package main

import (
	"bytes"
	"encoding/json"
	"net/http"
)

func verifyCaptcha(token, clientIP, secret string) (bool, int) {
	body, _ := json.Marshal(map[string]string{
		"token": token,
		"ip":    clientIP,
	})

	req, _ := http.NewRequest("POST", "http://localhost:3000/api/siteverify", bytes.NewBuffer(body))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", "Bearer "+secret)

	resp, err := http.DefaultClient.Do(req)
	if err != nil || resp.StatusCode != 200 {
		return false, 0
	}
	defer resp.Body.Close()

	var result struct {
		Success bool `json:"success"`
		Score   int  `json:"score"`
	}
	json.NewDecoder(resp.Body).Decode(&result)
	return result.Success, result.Score
}
